    let todos = [];
function addTodo(task) {
    if (task.trim() === '') {
        console.log('任务内容不能为空');
        return false;
    }
    const newTodo = {
        id: todos.length + 1,
        task: task,
        completed: false,
        createdAt: new Date().toLocaleString()
    };
    todos.push(newTodo);
    console.log(`任务已添加: ${task}`);
    return true;
}
function completeTodo(id) {
    const todo = todos.find(item => item.id === id);
    if (todo) {
        todo.completed = true;
        console.log(`任务已完成: ${todo.task}`);
    } else {
        console.log(`未找到ID为 ${id} 的任务`);
    }
}
function listTodos() {
    if (todos.length === 0) {
        console.log('暂无任务');
        return;
    }
    console.log('=== 待办列表 ===');
    todos.forEach(todo => {
        const status = todo.completed ? '✓' : '○';
        console.log(`[${status}] ${todo.id}: ${todo.task}`);
    });
}
addTodo('学习JavaScript');
addTodo('写代码示例');
listTodos();
completeTodo(1);
listTodos();
console.log(`总任务数: ${todos.length}`);
